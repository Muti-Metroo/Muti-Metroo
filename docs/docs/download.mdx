export const DownloadButton = ({href, children, primary}) => (
  <a
    href={href}
    className={`download-button ${primary ? 'download-button--primary' : ''}`}
    target="_blank"
    rel="noopener noreferrer">
    {children}
  </a>

export const DownloadButtonGroup = ({children}) => (
  <div className="download-button-group">
    {children}
  </div>
);

export const OSDetector = () => {
  useEffect(() => {
    const detectOS = () => {
      const ua = navigator.userAgent.toLowerCase();
      const platform = navigator.platform?.toLowerCase() || '';
      if (ua.includes('win')) return 'windows';
      if (ua.includes('mac') || platform.includes('mac')) return 'macos';
      if (ua.includes('linux') || platform.includes('linux')) return 'linux';
      return null;
    };

    const os = detectOS();
    if (!os) return;

    const tabMap = {
      'macos': 'macOS',
      'linux': 'Linux',
      'windows': 'Windows'
    };
    const tabLabel = tabMap[os];
    if (!tabLabel) return;

    setTimeout(() => {
      const tabs = document.querySelectorAll('[role="tab"]');
      tabs.forEach(tab => {
        if (tab.textContent === tabLabel) {
          tab.click();
        }
      });
    }, 100);

}, []);

return null;
};

### Installation

```bash
# Download for your architecture (example: Apple Silicon)
curl -L -o muti-metroo __MUTI_METROO_RELEASE__/muti-metroo-darwin-arm64

# Make executable and install
chmod +x muti-metroo
sudo mv muti-metroo /usr/local/bin/

# Verify installation
muti-metroo --version
```

:::note Gatekeeper Warning
On first run, macOS may show a security warning because the binary is not signed with a Developer ID certificate. To allow it:
1. Right-click the binary and select "Open", or
2. Go to System Preferences > Security & Privacy and click "Allow Anyway"
:::

:::tip DLL Mode
The DLL variant runs via `rundll32.exe` without a console window. Use `.exe` for normal operation; use `.dll` for background execution scenarios. The ARM64 DLL is a native PE image and works with the standard `rundll32.exe` in `System32` -- no emulation layer required. See [DLL Mode](/deployment/dll-mode) for details.
:::

### Installation

```powershell
# Download for your architecture (example: amd64)
curl.exe -L -o muti-metroo.exe __MUTI_METROO_RELEASE__/muti-metroo-windows-amd64.exe

# Move to Program Files (run as Administrator)
mkdir "C:\Program Files\muti-metroo" -ErrorAction SilentlyContinue
move muti-metroo.exe "C:\Program Files\muti-metroo\"

# Add to PATH (run as Administrator, or add manually via System Properties)
$path = [Environment]::GetEnvironmentVariable("Path", "Machine")
[Environment]::SetEnvironmentVariable("Path", "$path;C:\Program Files\muti-metroo", "Machine")

# Verify installation (restart terminal first)
muti-metroo.exe --version
```

## Verify Your Download

Verify your download with SHA256 checksums:

<DownloadButtonGroup>
  <DownloadButton href="__MUTI_METROO_RELEASE__/checksums.txt">Download Checksums</DownloadButton>
</DownloadButtonGroup>

<Tabs groupId="verify-os">
  <TabItem value="unix" label="Linux / macOS" default>

```bash
# Download checksums
curl -L -o checksums.txt __MUTI_METROO_RELEASE__/checksums.txt

# Verify (example for linux-amd64)
sha256sum -c checksums.txt --ignore-missing
```

  </TabItem>
  <TabItem value="windows" label="Windows">

```powershell
# Get the expected hash from checksums.txt (example: .exe)
$expected = (Get-Content checksums.txt | Select-String "muti-metroo-windows-amd64.exe").ToString().Split()[0]

# Calculate actual hash
$actual = (Get-FileHash muti-metroo-windows-amd64.exe -Algorithm SHA256).Hash.ToLower()

# Compare
if ($expected -eq $actual) { "OK" } else { "MISMATCH" }

# For DLL verification, use muti-metroo-windows-amd64.dll or muti-metroo-windows-arm64.dll instead
```
