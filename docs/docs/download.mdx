export const DownloadButton = ({href, children, primary}) => (
  <a
    href={href}
    className={`download-button ${primary ? 'download-button--primary' : ''}`}
    target="_blank"

    {children}

export const DownloadButtonGroup = ({children}) => (
  <div className="download-button-group">

  </div>

export const OSDetector = () => {
  useEffect(() => {
    const detectOS = () => {
      const ua = navigator.userAgent.toLowerCase();
      const platform = navigator.platform?.toLowerCase() || '';
      if (ua.includes('win')) return 'windows';
      if (ua.includes('mac') || platform.includes('mac')) return 'macos';
      if (ua.includes('linux') || platform.includes('linux')) return 'linux';
      return null;
    };

    const os = detectOS();
    if (!os) return;

    const tabMap = {
      'macos': 'macOS',
      'linux': 'Linux',
      'windows': 'Windows'
    };
    const tabLabel = tabMap[os];
    if (!tabLabel) return;

    setTimeout(() => {
      const tabs = document.querySelectorAll('[role="tab"]');
      tabs.forEach(tab => {
        if (tab.textContent === tabLabel) {
          tab.click();
        }
      });
    }, 100);

}, []);

return null;
};

### Install on macOS

This example installs the Apple Silicon binary. For Intel Macs, use `muti-metroo-darwin-amd64`.

```bash
curl -L -O https://github.com/Muti-Metroo/Muti-Metroo/releases/latest/download/muti-metroo-darwin-arm64
chmod +x muti-metroo-darwin-arm64
sudo mv muti-metroo-darwin-arm64 /usr/local/bin/muti-metroo
muti-metroo --version
```

:::note[Gatekeeper]
The macOS binaries are not signed with an Apple Developer ID. A file downloaded with `curl` runs directly. A file downloaded with a web browser is quarantined, and macOS refuses to run it. Remove the quarantine attribute before running it:

```bash
xattr -d com.apple.quarantine /usr/local/bin/muti-metroo
```
:::

:::tip[DLL mode]
The DLL runs the agent through `rundll32.exe` without a console window. Use the `.exe` for normal operation and the `.dll` for background execution. The ARM64 DLL is a native ARM64 image and runs with the standard `rundll32.exe` in `System32`. See [DLL Mode](/deployment/dll-mode).

### Install on Windows

This example installs the x86_64 executable. For ARM64, use `muti-metroo-windows-arm64.exe`. Run PowerShell as Administrator.

```powershell
curl.exe -L -O https://github.com/Muti-Metroo/Muti-Metroo/releases/latest/download/muti-metroo-windows-amd64.exe

# Install into Program Files
New-Item -ItemType Directory -Force "C:\Program Files\muti-metroo"
Move-Item muti-metroo-windows-amd64.exe "C:\Program Files\muti-metroo\muti-metroo.exe"

# Add the directory to the system PATH
$path = [Environment]::GetEnvironmentVariable("Path", "Machine")
[Environment]::SetEnvironmentVariable("Path", "$path;C:\Program Files\muti-metroo", "Machine")
```

Open a new terminal so the PATH change takes effect, then check the installation:

```powershell
muti-metroo.exe --version
```

## Verify your download

Each release publishes SHA-256 checksums for all assets in `checksums.txt`. Check the downloaded file before you rename or move it, because the checksum file lists the original asset names.

<DownloadButtonGroup>
  <DownloadButton href="https://github.com/Muti-Metroo/Muti-Metroo/releases/latest/download/checksums.txt">Download Checksums</DownloadButton>
</DownloadButtonGroup>

<Tabs groupId="verify-os">
  <TabItem value="unix" label="Linux / macOS" default>

Run this in the directory that holds the downloaded asset:

```bash
curl -L -O https://github.com/Muti-Metroo/Muti-Metroo/releases/latest/download/checksums.txt
sha256sum -c checksums.txt --ignore-missing
```

```text
muti-metroo-linux-amd64: OK
```

On macOS versions without `sha256sum`, use `shasum -a 256 -c checksums.txt --ignore-missing`.

  </TabItem>
  <TabItem value="windows" label="Windows">

```powershell
$file = "muti-metroo-windows-amd64.exe"
$expected = (Select-String -Path checksums.txt -Pattern " $file$").Line.Split()[0]
$actual = (Get-FileHash $file -Algorithm SHA256).Hash.ToLower()
if ($expected -eq $actual) { "OK" } else { "MISMATCH" }
```

Set `$file` to the name of the asset you downloaded, for example `muti-metroo-windows-arm64.dll`.

## Mutiauk

Mutiauk runs on the operator's workstation. It creates a TUN interface, installs the mesh's routes, and forwards IP traffic through an ingress agent's SOCKS5 listener, so applications need no proxy settings. It requires root or Administrator. Release binaries are published for Linux; see the [Mutiauk guide](/mutiauk) for details.
